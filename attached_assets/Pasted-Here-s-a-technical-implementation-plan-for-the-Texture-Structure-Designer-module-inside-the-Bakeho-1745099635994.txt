3. Texture & Structure Panel UI

/components/TextureStructurePanel.jsx

export default function TextureStructurePanel() {
  const { textureSettings, setTextureSettings } = useRecipe();

  const handleChange = (field, value) => {
    setTextureSettings(prev => ({ ...prev, [field]: value }));
  };

  return (
    <section className="bg-white p-6 rounded-md shadow grid grid-cols-1 md:grid-cols-2 gap-6">
      <h2 className="col-span-full text-xl font-semibold text-stone-800">Texture & Structure</h2>

      <TextureSlider label="Crumb Openness" value={textureSettings.crumbOpenness} onChange={val => handleChange('crumbOpenness', val)} />
      <TextureSlider label="Crumb Elasticity" value={textureSettings.elasticity} onChange={val => handleChange('elasticity', val)} />
      <TextureSlider label="Crust Thickness" value={textureSettings.crustThickness} onChange={val => handleChange('crustThickness', val)} />
      <TextureSlider label="Crust Hardness" value={textureSettings.crustHardness} onChange={val => handleChange('crustHardness', val)} />
      <TextureSlider label="Chew Level" value={textureSettings.chewiness} onChange={val => handleChange('chewiness', val)} />

      <div className="col-span-full">
        <label className="block font-medium mb-1">Rise Profile</label>
        <div className="flex gap-2">
          {['Domed', 'Flat', 'Rustic'].map(profile => (
            <button
              key={profile}
              onClick={() => handleChange('riseProfile', profile)}
              className={`px-4 py-2 border rounded ${textureSettings.riseProfile === profile ? 'bg-amber-600 text-white' : 'bg-white'}`}
            >
              {profile}
            </button>
          ))}
        </div>
      </div>
    </section>
  );
}

4. TextureSlider Component

export default function TextureSlider({ label, value, onChange }) {
  return (
    <div>
      <label className="block font-medium mb-2">{label}</label>
      <input
        type="range"
        min="0"
        max="100"
        value={value}
        onChange={e => onChange(parseInt(e.target.value))}
        className="w-full"
      />
      <div className="text-sm text-stone-600 mt-1">{value}/100</div>
    </div>
  );
}

5. Global State Context for Recipe Logic

/context/RecipeContext.jsx

const RecipeContext = createContext();

export function RecipeProvider({ children }) {
  const [textureSettings, setTextureSettings] = useState({
    crumbOpenness: 60,
    elasticity: 50,
    crustThickness: 40,
    crustHardness: 30,
    chewiness: 50,
    riseProfile: 'Rustic'
  });

  return (
    <RecipeContext.Provider value={{ textureSettings, setTextureSettings }}>
      {children}
    </RecipeContext.Provider>
  );
}

export const useRecipe = () => useContext(RecipeContext);

6. Mapping Texture to Recipe Adjustments

/utils/textureMapping.js

export function getHydrationLevel({ crumbOpenness, elasticity }) {
  if (crumbOpenness > 70 && elasticity > 60) return 78;
  if (crumbOpenness < 40) return 65;
  return 72;
}

export function adjustCrustSettings({ crustThickness, crustHardness }) {
  return {
    bakeTemp: 475,
    bakeTime: crustThickness > 50 ? 45 : 30,
    steamRequired: crustHardness > 50,
  };
}
