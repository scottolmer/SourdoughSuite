function adjustHydration(recipe, targetHydration) {
  // Current hydration
  const currentHydration = recipe.hydration;
  
  // Calculate total flour weight (baker's percentage base)
  const totalFlourWeight = recipe.ingredients
    .filter(i => i.category === 'flour')
    .reduce((sum, flour) => sum + flour.amount, 0);
  
  // Current water weight
  const waterIngredient = recipe.ingredients.find(i => i.name.toLowerCase() === 'water');
  const currentWaterWeight = waterIngredient ? waterIngredient.amount : 0;
  
  // Calculate new water weight
  const newWaterWeight = (totalFlourWeight * targetHydration) / 100;
  
  // Update water ingredient
  if (waterIngredient) {
    waterIngredient.amount = Math.round(newWaterWeight);
    waterIngredient.percentage = targetHydration;
  } else {
    // Add water if not present (unlikely scenario)
    recipe.ingredients.push({
      name: 'Water',
      amount: Math.round(newWaterWeight),
      unit: 'g',
      percentage: targetHydration,
      category: 'liquid'
    });
  }
  
  // Update recipe hydration
  recipe.hydration = targetHydration;
  
  // Update total dough weight
  updateTotalDoughWeight(recipe);
  
  // Adjust process for new hydration
  adjustProcessForHydration(recipe, currentHydration, targetHydration);
}

function adjustProcessForHydration(recipe, oldHydration, newHydration) {
  // Hydration change impacts several process aspects
  const hydrationIncrease = newHydration > oldHydration;
  
  // Adjust applicable process steps
  recipe.process.forEach(step => {
    // Adjust autolyse duration if present
    if (step.name === 'Autolyse') {
      if (hydrationIncrease) {
        // Higher hydration = shorter autolyse needed
        step.duration = Math.max(20, step.duration - 10);
      } else {
        // Lower hydration = longer autolyse beneficial
        step.duration += 10;
      }
    }
    
    // Adjust bulk fermentation handling
    if (step.name === 'Bulk Fermentation') {
      if (hydrationIncrease && newHydration > 78) {
        // Update technique guidance for very wet dough
        step.description = step.description.replace(
          /stretch and folds/i, 
          'coil folds or lamination'
        );
        
        // Update critical factors
        step.criticalFactors = step.criticalFactors.filter(f => 
          !f.includes('stretch and fold')
        );
        step.criticalFactors.push('gentle dough handling to maintain gas');
        step.criticalFactors.push('adequate dough strength development');
      }
    }
    
    // Adjust folding technique if present
    if (step.name.includes('Fold') || step.description.includes('fold')) {
      if (hydrationIncrease && newHydration > 78) {
        // Switch to coil folds for high hydration
        step.name = step.name.replace(/Stretch and Fold/i, 'Coil Fold');
        step.description = step.description.replace(
          /stretch and fold/i,
          'coil fold'
        );
      } else if (!hydrationIncrease && newHydration < 70) {
        // Switch to stretch and folds for lower hydration
        step.name = step.name.replace(/Coil Fold/i, 'Stretch and Fold');
        step.description = step.description.replace(
          /coil fold/i,
          'stretch and fold'
        );
      }
    }
    
    // Adjust shaping guidance
    if (step.name === 'Final Shape') {
      if (hydrationIncrease && newHydration > 78) {
        step.description = `For this high-hydration dough (${newHydration}%), use a gentle approach with minimal flour. Focus on building tension through folding rather than aggressive shaping.`;
      } else if (!hydrationIncrease && newHydration < 70) {
        step.description = `Shape the dough (${newHydration}% hydration) by creating good surface tension. This moderate hydration allows for standard shaping techniques with minimal sticking.`;
      }
    }
  });
}
