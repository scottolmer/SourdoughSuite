```javascript
function ParameterGauge({ label, value, status, ideal, min, max, step = 1, unit = '%' }) {
  // Status colors
  const statusColors = {
    'pass': '#10B981', // green
    'warning': '#F59E0B', // amber
    'error': '#EF4444' // red
  };
  
  const color = statusColors[status] || statusColors.warning;
  
  // Calculate percentage position on the gauge
  const percentage = Math.min(Math.max(((value - min) / (max - min)) * 100, 0), 100);
  const idealPercentage = Math.min(Math.max(((ideal - min) / (max - min)) * 100, 0), 100);
  
  return (
    <div className="parameter-gauge p-4 bg-white rounded-lg border">
      <div className="flex items-center justify-between mb-2">
        <span className="text-sm font-medium text-gray-700">{label}</span>
        <span className="text-xl font-semibold" style={{ color }}>
          {typeof value === 'number' ? value.toFixed(step < 1 ? 1 : 0) : value}{unit}
        </span>
      </div>
      
      <div className="gauge-track h-2 bg-gray-200 rounded-full relative mb-1">
        {/* Ideal marker */}
        <div 
          className="ideal-marker absolute w-1 h-4 bg-gray-500 rounded-full -mt-1" 
          style={{ left: `${idealPercentage}%` }}
          title={`Ideal: ${ideal}${unit}`}
        />
        
        {/* Value indicator */}
        <div 
          className="gauge-fill h-full rounded-full" 
          style={{ 
            width: `${percentage}%`, 
            backgroundColor: color 
          }} 
        />
      </div>
      
      <div className="gauge-labels flex justify-between text-xs text-gray-500">
        <span>{min}{unit}</span>
        <span>{max}{unit}</span>
      </div>
    </div>
  );
}
```

```javascript
// Identify differences between original and optimized recipes
function hasIngredientChanged(ingredient, comparisonIngredients) {
  const comparisonIngredient = comparisonIngredients.find(i => i.name === ingredient.name);
  
  if (!comparisonIngredient) return true; // New ingredient
  
  // Check if any property has changed
  return ingredient.amount !== comparisonIngredient.amount || 
         ingredient.percentage !== comparisonIngredient.percentage;
}

function hasProcessStepChanged(step, comparisonProcess, index) {
  // If index is out of bounds, it's a new step
  if (index >= comparisonProcess.length) return true;
  
  const comparisonStep = comparisonProcess[index];
  
  // Check if step name is different (likely a different step)
  if (step.name !== comparisonStep.name) return true;
  
  // Check if key properties have changed
  return step.duration !== comparisonStep.duration || 
         step.description !== comparisonStep.description ||
         step.temperature !== comparisonStep.temperature;
}

function identifyProcessChanges(originalRecipe, optimizedRecipe) {
  const changes = [];
  
  // Check for added or removed steps
  if (originalRecipe.process.length !== optimizedRecipe.process.length) {
    if (originalRecipe.process.length < optimizedRecipe.process.length) {
      changes.push('Added missing process steps');
    } else {
      changes.push('Removed unnecessary steps');
    }
  }
  
  // Check for bulk fermentation changes
  const originalBulk = originalRecipe.process.find(s => s.name.includes('Bulk Fermentation'));
  const optimizedBulk = optimizedRecipe.process.find(s => s.name.includes('Bulk Fermentation'));
  
  if (originalBulk && optimizedBulk && originalBulk.duration !== optimizedBulk.duration) {
    const diff = optimizedBulk.duration - originalBulk.duration;
    changes.push(`${diff > 0 ? 'Extended' : 'Reduced'} bulk fermentation by ${Math.abs(diff)} minutes`);
  }
  
  // Check for autolyse changes
  const hasOriginalAutolyse = originalRecipe.process.some(s => s.name.includes('Autolyse'));
  const hasOptimizedAutolyse = optimizedRecipe.process.some(s => s.name.includes('Autolyse'));
  
  if (!hasOriginalAutolyse && hasOptimizedAutolyse) {
    changes.push('Added autolyse step for better development');
  }
  
  // Check for technique changes
  const originalTechniques = extractTechniques(originalRecipe);
  const optimizedTechniques = extractTechniques(optimizedRecipe);
  
  const uniqueOptimizedTechniques = optimizedTechniques.filter(t => !originalTechniques.includes(t));
  
  if (uniqueOptimizedTechniques.length > 0) {
    changes.push(`Improved techniques: ${uniqueOptimizedTechniques.join(', ')}`);
  }
  
  return changes;
}

function extractTechniques(recipe) {
  const techniques = [];
  
  recipe.process.forEach(step => {
    // Extract technique mentions from descriptions
    const techniquePhrases = [
      'stretch and fold', 'coil fold', 'lamination', 
      'slap and fold', 'cold fermentation', 'autolyse'
    ];
    
    techniquePhrases.forEach(phrase => {
      if (step.description.toLowerCase().includes(phrase)) {
        techniques.push(phrase);
      }
    });
  });
  
  return [...new Set(techniques)]; // Remove duplicates
}

function findIngredientPercentage(recipe, category) {
  const ingredient = recipe.ingredients.find(i => i.category === category);
  return ingredient ? ingredient.percentage : 0;
}

function calculateTotalTime(recipe) {
  const totalMinutes = recipe.process.reduce((sum, step) => sum + step.duration, 0);
  const hours = Math.floor(totalMinutes / 60);
  const minutes = totalMinutes % 60;
  
  return `${hours}h ${minutes}m`;
}

function formatDuration(minutes) {
  if (minutes < 60) {
    return `${minutes} min`;
  }
  
  const hours = Math.floor(minutes / 60);
  const remainingMinutes = minutes % 60;
  
  if (remainingMinutes === 0) {
    return `${hours} hr`;
  }
  
  return `${hours} hr ${remainingMinutes} min`;
}
```

```javascript
function RecipeValidatorResultsPage({ validationResults, originalRecipe, optimizedRecipe }) {
  const [currentRecipe, setCurrentRecipe] = useState(originalRecipe);
```

### Key Implementation Considerations

1. **State Management**: The implementation keeps track of which recipe version is being viewed and which fixes have been applied.

2. **Fix Application Logic**: The `applySpecificFix` function would need a detailed implementation that surgically applies a single fix from the optimized recipe to the current recipe.

3. **Performance Optimization**: For complex recipes, comparison calculations should be memoized to avoid repeated deep comparison operations.

4. **Accessibility**: The implementation includes appropriate ARIA attributes and keyboard navigation, though I've omitted some for brevity.

5. **Mobile Responsiveness**: The split view should collapse to a tabbed interface on smaller screens.
